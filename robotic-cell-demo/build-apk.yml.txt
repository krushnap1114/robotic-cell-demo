name: Build Android APK

on:
  workflow_dispatch:        # lets you press "Run workflow" by hand
  push:
    branches: [main, master]

permissions:
  contents: write           # needed to publish the APK on the Releases page

jobs:
  apk:
    runs-on: ubuntu-latest
    timeout-minutes: 40
    steps:
      - name: Get the code
        uses: actions/checkout@v4

      - name: Install Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - name: Install Flutter
        uses: subosito/flutter-action@v2
        with:
          channel: stable
          cache: true

      - name: Create the Android runner folder
        run: |
          flutter create --org in.uvklink --project-name robotic_cell_demo --platforms android _platform_tmp
          cp -R _platform_tmp/android ./android
          rm -rf _platform_tmp
          perl -pi -e 's/android:label="[^"]*"/android:label="Robotic Cell IO"/' android/app/src/main/AndroidManifest.xml

      - name: Get packages
        run: flutter pub get

      - name: Build the APK
        run: flutter build apk --release

      - name: Keep the APK as a download (Actions page)
        uses: actions/upload-artifact@v4
        with:
          name: robotic-cell-demo-apk
          path: build/app/outputs/flutter-apk/app-release.apk

      - name: Publish the APK on the Releases page
        uses: softprops/action-gh-release@v2
        with:
          tag_name: demo-${{ github.run_number }}
          name: Robotic Cell I/O demo (build ${{ github.run_number }})
          body: Demo app with simulated data only. Download app-release.apk on your Android phone and open it.
          files: build/app/outputs/flutter-apk/app-release.apk
